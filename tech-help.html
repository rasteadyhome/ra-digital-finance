const express = require('express'); const { OpenAI } = require('openai'); const app = express(); app.use(express.json()); app.use(express.static('.')); const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY }); app.post('/api/chat', async (req, res) => { try { const { messages } = req.body; const completion = await openai.chat.completions.create({ model: 'gpt-4o-mini', messages: messages, }); res.json(completion.choices[0].message); } catch (error) { console.error('API Error:', error); res.status(500).json({ error: error.message }); } }); const PORT = process.env.PORT || 3000; app.listen(PORT, () => { console.log(`Server running on port ${PORT}`); });
